cmake_minimum_required(VERSION 3.16)
project(balance_homie_g1_deploy LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Python3 3.8 REQUIRED COMPONENTS Interpreter)
find_package(Threads REQUIRED)
set(SDK "${CMAKE_CURRENT_SOURCE_DIR}/vendor/unitree_sdk2")
set(ARCH "${CMAKE_SYSTEM_PROCESSOR}")
if(ARCH STREQUAL "arm64")
  set(ARCH aarch64)
endif()
if(NOT ARCH STREQUAL "x86_64" AND NOT ARCH STREQUAL "aarch64")
  message(FATAL_ERROR "Only x86_64 and aarch64 are bundled")
endif()
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS config/deploy.json
  config/balance_pd.json config/homie_pd.json config/pushcart_upper_pd.json tools/generate_config.py
  python/hybrid_deploy/configuration.py)
execute_process(COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tools/generate_config.py"
  "${CMAKE_CURRENT_SOURCE_DIR}/config/deploy.json" "${CMAKE_CURRENT_BINARY_DIR}/generated_config.hpp"
  RESULT_VARIABLE CONFIG_RESULT)
if(NOT CONFIG_RESULT EQUAL 0)
  message(FATAL_ERROR "Invalid deployment config")
endif()
add_executable(g1_control cpp/g1_control.cpp cpp/dex3_type_support.cpp)
target_compile_options(g1_control PRIVATE -Wall -Wextra -Wpedantic)
target_include_directories(g1_control PRIVATE cpp "${CMAKE_CURRENT_BINARY_DIR}")
target_include_directories(g1_control SYSTEM PRIVATE
  "${SDK}/include" "${SDK}/thirdparty/include" "${SDK}/thirdparty/include/ddscxx")
target_link_libraries(g1_control PRIVATE "${SDK}/lib/${ARCH}/libunitree_sdk2.a"
  "${SDK}/thirdparty/lib/${ARCH}/libddscxx.so" "${SDK}/thirdparty/lib/${ARCH}/libddsc.so" Threads::Threads)
set_target_properties(g1_control PROPERTIES BUILD_RPATH "${SDK}/thirdparty/lib/${ARCH}")
enable_testing()
add_executable(test_command_trace tests/test_command_trace.cpp)
target_compile_options(test_command_trace PRIVATE -UNDEBUG)
target_include_directories(test_command_trace PRIVATE cpp)
add_test(NAME command_trace COMMAND test_command_trace)
add_executable(test_control_core tests/test_control_core.cpp)
target_compile_options(test_control_core PRIVATE -UNDEBUG)
target_include_directories(test_control_core PRIVATE cpp "${CMAKE_CURRENT_BINARY_DIR}")
add_test(NAME control_core COMMAND test_control_core)
add_executable(test_hand_core tests/test_hand_core.cpp)
target_compile_options(test_hand_core PRIVATE -UNDEBUG)
target_include_directories(test_hand_core PRIVATE cpp)
add_test(NAME hand_core COMMAND test_hand_core)
add_executable(test_hand_recovery tests/test_hand_recovery.cpp)
target_compile_options(test_hand_recovery PRIVATE -UNDEBUG)
target_include_directories(test_hand_recovery PRIVATE cpp)
add_test(NAME hand_recovery COMMAND test_hand_recovery)
add_executable(test_hand_dds tests/test_hand_dds.cpp cpp/dex3_type_support.cpp)
target_include_directories(test_hand_dds PRIVATE cpp)
target_include_directories(test_hand_dds SYSTEM PRIVATE
  "${SDK}/include" "${SDK}/thirdparty/include" "${SDK}/thirdparty/include/ddscxx")
target_link_libraries(test_hand_dds PRIVATE
  "${SDK}/lib/${ARCH}/libunitree_sdk2.a"
  "${SDK}/thirdparty/lib/${ARCH}/libddscxx.so" "${SDK}/thirdparty/lib/${ARCH}/libddsc.so" Threads::Threads)
set_target_properties(test_hand_dds PROPERTIES BUILD_RPATH "${SDK}/thirdparty/lib/${ARCH}")
add_test(NAME hand_dds COMMAND test_hand_dds)
